Hmm, for import, simpler: normalizeState handles both load and import. For items with missing roomId: create an "Unsorted" room if needed. Let me do that in normalizeState:

```js
function normalizeState(data) {
  const rooms = [];
  const seenIds = new Set();
  if (Array.isArray(data?.rooms)) {
    for (const r of data.rooms) {
      if (!r || typeof r.name !== "string" || !r.name.trim()) continue;
      const id = r.id != null ? String(r.id) : makeId();
      if (seenIds.has(id)) continue;
      seenIds.add(id);
      rooms.push({ id, name: r.name.trim() });
    }
  }
  const items = [];
  if (Array.isArray(data?.items)) {
    for (const i of data.items) {
      if (!i || typeof i.name !== "string" || !i.name.trim()) continue;
      const id = i.id != null ? String(i.id) : makeId();
      const roomId = seenIds.has(i.roomId) ? String(i.roomId) : null;
      items.push({
        id,
        name: i.name.trim(),
        quantity: Math.max(0, Math.floor(Number(i.quantity) || 0)),
        category: CATEGORIES.includes(i.category) ? i.category : "Other",
        roomId,
        notes: typeof i.notes === "string" ? i.notes : "",
        createdAt: Number.isFinite(Number(i.createdAt)) ? Number(i.createdAt) : Date.now(),
      });
    }
  }
  // Attach orphan items to Unsorted
  if (items.some(i => i.roomId === null)) {
    const existing = rooms.find(r => r.name === UNSORTED_ROOM);
    const target = existing ?? { id: makeId(), name: UNSORTED_ROOM };
    if (!existing) { rooms.push(target); seenIds.add(target.id); }
    for (const i of items) if (i.roomId === null) i.roomId = target.id;
  }
  return { rooms, items };
}
```

```js
function makeId() {
  if (typeof crypto !== "undefined" && crypto.randomUUID) return crypto.randomUUID();
  return "id-" + Date.now().toString(36) + "-" + Math.random().toString(36).slice(2, 10);
}
```

```js
function startRename(btn) {
  const li = btn.closest("li");
  const nameEl = li.querySelector(".room-name");
  const input = document.createElement("input");
  input.type = "text";
  input.value = nameEl.textContent;
  input.className = "rename-input";
  input.setAttribute("aria-label", "Rename room");
  nameEl.replaceWith(input);
  input.focus();
  input.select();
  const commit = (cancel) => {
    const value = input.value.trim();
    input.remove();
    if (cancel) { renderRooms(); return; }
    if (!value) { toast("Room name can't be empty.", { tone: "warn" }); renderRooms(); return; }
    const duplicate = state.rooms.some(r => r.id !== li.dataset.roomId && r.name.toLowerCase() === value.toLowerCase());
    if (duplicate) { toast("A room with that name already exists.", { tone: "warn" }); renderRooms(); return; }
    const room = state.rooms.find(r => r.id === li.dataset.roomId);
    room.name = value;
    saveState();
    renderAll();
    toast(`Renamed room to "${value}".`);
  };
  input.addEventListener("keydown", (e) => {
    if (e.key === "Enter") commit(false);
    else if (e.key === "Escape") commit(true);
  });
  input.addEventListener("blur", () => { if (input.isConnected) commit(false); });
}
```

```js
  e.preventDefault();
  const name = roomNameInput.value.trim();
  if (!name) { roomNameInput.focus(); return; }
  if (state.rooms.some(r => r.name.toLowerCase() === name.toLowerCase())) {
    toast("A room named \"" + name + "\" already exists.", { tone: "warn" });
    return;
  }
  state.rooms.push({ id: makeId(), name });
  saveState();
  renderAll();
  roomNameInput.value = "";
  roomNameInput.focus();
  toast("Added room \"" + name + "\".");
});
```
