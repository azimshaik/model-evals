<script>
/**
 * Home Inventory Manager
 * Single file application logic.
 */
const app = {
    data: {
        rooms: [],
        items: []
    },
    currentRoomId: null,
    categories: ['Food', 'Tools', 'Documents', 'Electronics', 'Cleaning', 'Other'],

    init() {
        this.loadData();
        this.renderRooms();
        this.renderItems();
        this.setupEventListeners();
    },

    // --- Persistence ---
    loadData() {
        const storedRooms = localStorage.getItem('homeInventory_rooms');
        const storedItems = localStorage.getItem('homeInventory_items');

        if (storedRooms) {
            this.data.rooms = JSON.parse(storedRooms);
        } else {
            this.data.rooms = ['Kitchen', 'Garage', 'Study'];
        }

        if (storedItems) {
            this.data.items = JSON.parse(storedItems);
        } else {
            this.data.items = [];
        }

        this.currentRoomId = this.data.rooms.length > 0 ? this.data.rooms[0] : null;
    },

    saveData() {
        localStorage.setItem('homeInventory_rooms', JSON.stringify(this.data.rooms));
        localStorage.setItem('homeInventory_items', JSON.stringify(this.data.items));
    },

    // --- Rendering ---
    renderRooms() {
        const container = document.getElementById('room-tabs');
        container.innerHTML = '';
        
        if (this.data.rooms.length === 0) {
            container.innerHTML = '<span style="color:#64748b; padding: 0.5rem;">No rooms defined. Add one to get started.</span>';
            return;
        }

        this.data.rooms.forEach(room => {
            const btn = document.createElement('button');
            btn.className = `tab-btn ${this.currentRoomId === room ? 'active' : ''}`;
            btn.textContent = room;
            btn.onclick = () => this.switchRoom(room);
            container.appendChild(btn);
        });
    },

    renderItems() {
        const tbody = document.getElementById('items-tbody');
        tbody.innerHTML = '';

        // Filter Logic
        const searchTerm = document.getElementById('search-input').value.toLowerCase();
        const filterRoom = document.getElementById('filter-room').value;
        const filterCat = document.getElementById('filter-category').value;

        let filtered = this.data.items.filter(item => {
            const matchSearch = item.name.toLowerCase().includes(searchTerm) || 
                                (item.notes && item.notes.toLowerCase().includes(searchTerm));
            const matchRoom = filterRoom ? item.room === filterRoom : true;
            const matchCat = filterCat ? item.category === filterCat : true;
            return matchSearch && matchRoom && matchCat;
        });

        if (filtered.length === 0) {
            tbody.innerHTML = '<tr><td colspan="6" style="text-align:center; padding: 2rem; color: #64748b;">No items found</td></tr>';
            return;
        }

        filtered.forEach(item => {
            const tr = document.createElement('tr');
            const isLowStock = item.quantity <= 2;
            
            tr.innerHTML = `
                <td data-label="Name">${item.name}</td>
                <td data-label="Qty" class="${isLowStock ? 'low-stock' : ''}">${item.quantity}</td>
                <td data-label="Category"><span class="badge">${item.category}</span></td>
                <td data-label="Room">${item.room}</td>
                <td data-label="Notes">${item.notes || '-'}</td>
                <td data-label="Actions">
                    <div class="item-actions">
                        <button onclick="app.editItem('${item.id}')" style="padding: 2px 8px;">Edit</button>
                        <button onclick="app.deleteItem('${item.id}')" class="danger" style="padding: 2px 8px;">Del</button>
                    </div>
                </td>
            `;
            tbody.appendChild(tr);
        });
    },

    // --- Actions ---
    switchRoom(roomName) {
        this.currentRoomId = roomName;
        this.renderRooms();
        this.filterItems(); // Re-render items based on new room
    },

    filterItems() {
        this.renderItems();
    },

    openModal(type) {
        const overlay = document.getElementById('modal-overlay');
        const title = document.getElementById('modal-title');
        const form = document.getElementById('modal-form');
        
        form.reset();
        document.getElementById('item-id').value = '';
        document.getElementById('room-id').value = '';

        if (type === 'room') {
            title.textContent = 'Add Room';
            form.querySelector('#inp-name').setAttribute('id', 'inp-room-name');
            form.querySelector('#inp-name').setAttribute('placeholder', 'Room Name');
        } else {
            title.textContent = 'Add Item';
            form.querySelector('#inp-name').setAttribute('id', 'inp-name');
            form.querySelector('#inp-name').setAttribute('placeholder', 'Item Name');
            
            // Populate Room Select
            const roomSelect = document.getElementById('inp-room');
            roomSelect.innerHTML = '';
            this.data.rooms.forEach(r => {
                const opt = document.createElement('option');
                opt.value = r;
                opt.textContent = r;
                roomSelect.appendChild(opt);
            });
        }
        
        overlay.classList.add('open');
    },

    closeModal() {
        document.getElementById('modal-overlay').classList.remove('open');
    },

    handleFormSubmit(e) {
        e.preventDefault();
        
        const id = document.getElementById('item-id').value;
        const name = document.getElementById('inp-name').value;
        const qty = parseInt(document.getElementById('inp-qty').value);
        const cat = document.getElementById('inp-category').value;
        const room = document.getElementById('inp-room').value;
        const notes = document.getElementById('inp-notes').value;

        if (id) {
            // Edit
            const index = this.data.items.findIndex(i => i.id === id);
            if (index > -1) {
                this.data.items[index] = { ...this.data.items[index], name, quantity: qty, category: cat, room, notes };
                this.showToast('Item updated');
            }
        } else {
            // Add Room
            if (name.trim() === '') return;
            this.data.rooms.push(name.trim());
            this.showToast('Room added');
        }

        // Add Item
        if (name.trim() !== '' && !isNaN(qty)) {
            const newItem = {
                id: Date.now().toString(),
                name,
                quantity: qty,
                category: cat,
                room,
                notes
            };
            this.data.items.push(newItem);
            this.showToast('Item added');
        }

        this.saveData();
        this.closeModal();
        this.renderRooms();
        this.renderItems();
    },

    editItem(id) {
        const item = this.data.items.find(i => i.id === id);
        if (!item) return;

        document.getElementById('item-id').value = item.id;
        document.getElementById('inp-name').value = item.name;
        document.getElementById('inp-qty').value = item.quantity;
        document.getElementById('inp-category').value = item.category;
        document.getElementById('inp-room').value = item.room;
        document.getElementById('inp-notes').value = item.notes || '';
        
        this.openModal('item');
    },

    deleteItem(id) {
        if (!confirm('Delete this item?')) return;
        
        this.data.items = this.data.items.filter(i => i.id !== id);
        this.saveData();
        this.renderItems();
        this.showToast('Item deleted');
    },

    // --- Import / Export ---
    exportData() {
        const dataStr = JSON.stringify(this.data);
        const blob = new Blob([dataStr], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'home-inventory-backup.json';
        a.click();
        URL.revokeObjectURL(url);
    },

    importData(input) {
        const file = input.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = (e) => {
            try {
                const json = JSON.parse(e.target.result);
                if (json.rooms && Array.isArray(json.rooms) && json.items && Array.isArray(json.items)) {
                    this.data = json;
                    this.saveData();
                    this.renderRooms();
                    this.renderItems();
                    this.showToast('Data imported successfully');
                } else {
                    alert('Invalid file format');
                }
            } catch (err) {
                alert('Error parsing JSON file');
            }
        };
        reader.readAsText(file);
        input.value = ''; // Reset input
    },

    loadDemo() {
        if (!confirm('This will overwrite your current data with demo data. Continue?')) return;

        this.data = {
            rooms: ['Kitchen', 'Garage', 'Study', 'Bedroom'],
            items: [
                { id: '1', name: 'Milk', quantity: 1, category: 'Food', room: 'Kitchen', notes: 'Needs restocking' },
                { id: '2', name: 'Wrench Set', quantity: 2, category: 'Tools', room: 'Garage', notes: 'Used for bikes' },
                { id: '3', name: 'USB Cable', quantity: 5, category: 'Electronics', room: 'Study', notes: 'For laptop' },
                { id: '4', name: 'Paperclips', quantity: 10, category: 'Documents', room: 'Study', notes: '' },
                { id: '5', name: 'Dish Soap', quantity: 0, category: 'Cleaning', room: 'Kitchen', notes: 'Empty!' },
                { id: '6', name: 'Winter Jacket', quantity: 1, category: 'Other', room: 'Bedroom', notes: 'In closet' }
            ]
        };
        this.saveData();
        this.renderRooms();
        this.renderItems();
        this.showToast('Demo data loaded');
    },

    showToast(msg) {
        const t = document.getElementById('toast');
        t.textContent = msg;
        t.classList.add('show');
        setTimeout(() => t.classList.remove('show'), 3000);
    },

    setupEventListeners() {
        // Populate Room Select in modal
        const roomSelect = document.getElementById('inp-room');
        if (roomSelect) {
            roomSelect.innerHTML = '';
            this.data.rooms.forEach(r => {
                const opt = document.createElement('option');
                opt.value = r;
                opt.textContent = r;
                roomSelect.appendChild(opt);
            });
        }
    }
};

// Start App
document.addEventListener('DOMContentLoaded', () => {
    app.init();
});
</script>
