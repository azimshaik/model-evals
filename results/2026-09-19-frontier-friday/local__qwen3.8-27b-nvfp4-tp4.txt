    type = type || "info";

    el.className = "toast " + type;
    el.setAttribute("role", "status");
    el.innerHTML = (ICONS[type] || ICONS.info) + "<span>" + escapeHtml(message) + "</span>";
    toastsEl.appendChild(el);
    requestAnimationFrame(function () { el.classList.add("show"); });
    setTimeout(function () {
      el.classList.remove("show");
      setTimeout(function () { el.remove(); }, 300);
    }, 2600);
  }

  /* ---------- Rendering ---------- */
  var roomListEl = document.getElementById("roomList");
  var roomCountEl = document.getElementById("roomCount");
  var itemListEl = document.getElementById("itemList");
  var emptyStateEl = document.getElementById("emptyState");
  var emptyTitleEl = document.getElementById("emptyTitle");
  var emptyTextEl = document.getElementById("emptyText");
  var emptyActionEl = document.getElementById("emptyAction");
  var visibleCountEl = document.getElementById("visibleCount");
  var statItemsEl = document.getElementById("statItems");
  var statRoomsEl = document.getElementById("statRooms");
  var statUnitsEl = document.getElementById("statUnits");
  var statLowEl = document.getElementById("statLow");
  var filterRoomEl = document.getElementById("filterRoom");
  var filterCategoryEl = document.getElementById("filterCategory");
  var clearFiltersEl = document.getElementById("clearFilters");

  function renderRooms() {
    roomCountEl.textContent = state.rooms.length;
    roomListEl.innerHTML = "";
    if (state.rooms.length === 0) {
      var li = document.createElement("li");
      li.style.cssText = "color:var(--ink-soft);font-size:.9rem;padding:4px 2px";
      li.textContent = "No rooms yet — add one below.";
      roomListEl.appendChild(li);
      return;
    }
    state.rooms.forEach(function (room) {
      var li = document.createElement("li");
      li.className = "room-row" + (filters.room === room.id ? " active" : "");
      li.innerHTML =
        '<button class="room-name" type="button" data-action="filter-room" data-id="' + room.id + '" title="Show items in ' + escapeHtml(room.name) + '">' + escapeHtml(room.name) + "</button>" +
        '<span class="room-count">' + itemCountForRoom(room.id) + "</span>" +
        '<button class="icon-btn" type="button" data-action="rename-room" data-id="' + room.id + '" aria-label="Rename ' + escapeHtml(room.name) + '" title="Rename">' +
          '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"><path d="M12 20h9"/><path d="M16.5 3.5a2.1 2.1 0 0 1 3 3L7 19l-4 1 1-4Z"/></svg>' +
        "</button>" +
        '<button class="icon-btn danger" type="button" data-action="delete-room" data-id="' + room.id + '" aria-label="Delete ' + escapeHtml(room.name) + '" title="Delete">' +
          '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round" stroke-linejoin="round"><path d="M3 6h18"/><path d="M8 6V4h8v2"/><path d="M6 6l1 14h10l1-14"/></svg>' +
        "</button>";
      roomListEl.appendChild(li);
    });
  }

  function renderRoomSelects() {
    var options = '<option value="all">All rooms</option>';
    state.rooms.forEach(function (r) {
      options += '<option value="' + r.id + '">' + escapeHtml(r.name) + "</option>";
    });
    filterRoomEl.innerHTML = options;
    filterRoomEl.value = filters.room;
    if (filterRoomEl.value !== filters.room) filters.room = "all";

    var formRoom = document.getElementById("itemRoom");
    var formOptions = state.rooms.length
      ? state.rooms.map(function (r) { return '<option value="' + r.id + '">' + escapeHtml(r.name) + "</option>"; }).join("")
      : '<option value="" disabled selected>No rooms yet</option>';
    formRoom.innerHTML = formOptions;
  }

  function renderCategorySelects() {
    var catOptions = CATEGORIES.map(function (c) { return '<option value="' + c + '">' + c + "</option>"; }).join("");
    document.getElementById("itemCategory").innerHTML = catOptions;
    filterCategoryEl.innerHTML = '<option value="all">All categories</option>' + catOptions;
    filterCategoryEl.value = filters.category;
  }

  function getVisibleItems() {
    var q = filters.query.trim().toLowerCase();
    return state.items.filter(function (item) {
      if (filters.room !== "all" && item.roomId !== filters.room) return false;
      if (filters.category !== "all" && item.category !== filters.category) return false;
      if (q) {
        var hay = (item.name + " " + (item.notes || "")).toLowerCase();
        if (hay.indexOf(q) === -1) return false;
      }
      return true;
    });
  }

  function renderItems() {
    var items = getVisibleItems();
    visibleCountEl.textContent = items.length;
    itemListEl.innerHTML = "";

    if (items.length === 0) {
      emptyStateEl.hidden = false;
      if (state.items.length === 0) {
        emptyTitleEl.textContent = "No items yet";
        emptyTextEl.textContent = "Add your first item, or load the demo data to explore.";
        emptyActionEl.hidden = false;
        emptyActionEl.textContent = "Load demo data";
      } else {
        emptyTitleEl.textContent = "No matching items";
        emptyTextEl.textContent = "Try a different search or clear the filters.";
        emptyActionEl.hidden = true;
      }
      return;
    }
    emptyStateEl.hidden = true;

    items.forEach(function (item) {
      var low = isLowStock(item);
      var li = document.createElement("li");
      li.className = "item" + (low ? " low" : "");
      li.innerHTML =
        '<div class="item-main">' +
          '<div class="item-top">' +
            '<span class="item-name">' + escapeHtml(item.name) + "</span>" +
            (low ? '<span class="chip low"><span class="dot"></span>Low stock</span>' : "") +
          "</div>" +
          (item.notes ? '<p class="item-notes">' + escapeHtml(item.notes) + "</p>" : "") +
          '<div class="item-meta">' +
            '<span class="chip cat"><span class="dot"></span>' + escapeHtml(item.category) + "</span>" +
            '<span class="chip room">' + escapeHtml(roomName(item.roomId)) + "</span>" +
            '<span class="chip">Qty ' + Number(item.quantity) + "</span>" +
          "</div>" +
        "</div>" +
        '<div class="item-actions">' +
          '<button class="btn btn-ghost btn-sm" type="button" data-action="edit-item" data-id="' + item.id + '" aria-label="Edit ' + escapeHtml(item.name) + '">Edit</button>' +
          '<button class="btn btn-danger btn-sm" type="button" data-action="delete-item" data-id="' + item.id + '" aria-label="Delete ' + escapeHtml(item.name) + '">Delete</button>' +
        "</div>";
      itemListEl.appendChild(li);
    });
  }

  function renderStats() {
    var totalUnits = state.items.reduce(function (sum, i) { return sum + Number(i.quantity); }, 0);
    var lowCount = state.items.filter(isLowStock).length;
    statItemsEl.textContent = state.items.length;
    statRoomsEl.textContent = state.rooms.length;
    statUnitsEl.textContent = totalUnits;
    statLowEl.textContent = lowCount;
  }

  function render() {
    renderRooms();
    renderRoomSelects();
    renderItems();
    renderStats();
    var hasFilter = filters.query.trim() !== "" || filters.room !== "all" || filters.category !== "all";
    clearFiltersEl.hidden = !hasFilter;
  }

  /* ---------- Item form ---------- */
  var itemForm = document.getElementById("itemForm");
  var itemEditIdEl = document.getElementById("itemEditId");
  var itemNameEl = document.getElementById("itemName");
  var itemQtyEl = document.getElementById("itemQty");
  var itemCategoryEl = document.getElementById("itemCategory");
  var itemRoomEl = document.getElementById("itemRoom");
  var itemNotesEl = document.getElementById("itemNotes");
  var itemSubmitEl = document.getElementById("itemSubmit");
  var itemCancelEl = document.getElementById("itemCancel");
  var itemFormTitle = document.querySelector(".item-form-title");

  function resetItemForm() {
    editingItemId = null;
    itemEditIdEl.value = "";
    itemForm.reset();
    itemQtyEl.value = "1";
    itemCategoryEl.value = CATEGORIES[0];
    itemSubmitEl.textContent = "Add item";
    itemCancelEl.hidden = true;
    itemFormTitle.textContent = "Add item";
  }

  function startEditItem(id) {
    var item = state.items.find(function (i) { return i.id === id; });
    if (!item) return;
    editingItemId = id;
    itemEditIdEl.value = id;
    itemNameEl.value = item.name;
    itemQtyEl.value = item.quantity;
    itemCategoryEl.value = item.category;
    itemRoomEl.value = item.roomId;
    itemNotesEl.value = item.notes || "";
    itemSubmitEl.textContent = "Save changes";
    itemCancelEl.hidden = false;
    itemFormTitle.textContent = "Edit item";
    itemNameEl.focus();
    if (window.innerWidth < 860) {
      itemForm.scrollIntoView({ behavior: "smooth", block: "center" });
    }
  }

  itemForm.addEventListener("submit", function (e) {
    e.preventDefault();
    if (state.rooms.length === 0) {
      toast("Create a room first", "danger");
      return;
    }
    var name = itemNameEl.value.trim();
    var quantity = Math.max(0, Math.floor(Number(itemQtyEl.value) || 0));
    var category = itemCategoryEl.value;
    var roomId = itemRoomEl.value;
    var notes = itemNotesEl.value.trim();
    if (!name) { itemNameEl.focus(); return; }

    if (editingItemId) {
      var item = state.items.find(function (i) { return i.id === editingItemId; });
      if (item) {
        item.name = name; item.quantity = quantity; item.category = category;
        item.roomId = roomId; item.notes = notes;
        toast("Item updated", "success");
      }
    } else {
      state.items.push({ id: uid(), name: name, quantity: quantity, category: category, roomId: roomId, notes: notes });
      toast("Item added", "success");
    }
    save();
    resetItemForm();
    render();
  });

  itemCancelEl.addEventListener("click", resetItemForm);

  /* ---------- Rooms ---------- */
  var roomForm = document.getElementById("roomForm");
  var roomInput = document.getElementById("roomInput");

  roomForm.addEventListener("submit", function (e) {
    e.preventDefault();
    var name = roomInput.value.trim();
    if (!name) { roomInput.focus(); return; }
    var exists = state.rooms.some(function (r) { return r.name.toLowerCase() === name.toLowerCase(); });
    if (exists) { toast("A room named “" + name + "” already exists", "info"); roomInput.focus(); return; }
    state.rooms.push({ id: uid(), name: name });
    save();
    roomInput.value = "";
    render();
    toast("Room added", "success");
  });

  function renameRoom(id) {
    var room = state.rooms.find(function (r) { return r.id === id; });
    if (!room) return;
    var next = window.prompt("Rename room", room.name);
    if (next === null) return;
    next = next.trim();
    if (!next) { toast("Room name can’t be empty", "danger"); return; }
    var dup = state.rooms.some(function (r) { return r.id !== id && r.name.toLowerCase() === next.toLowerCase(); });
    if (dup) { toast("A room named “" + next + "” already exists", "info"); return; }
    room.name = next;
    save();
    render();
    toast("Room renamed", "success");
  }

  function deleteRoom(id) {
    var room = state.rooms.find(function (r) { return r.id === id; });
    if (!room) return;
    var count = itemCountForRoom(id);
    var msg = count > 0
      ? "Delete “" + room.name + "” and its " + count + " item(s)? This can’t be undone."
      : "Delete “" + room.name + "”? This can’t be undone.";
    if (!window.confirm(msg)) return;
    state.rooms = state.rooms.filter(function (r) { return r.id !== id; });
    state.items = state.items.filter(function (i) { return i.roomId !== id; });
    if (filters.room === id) filters.room = "all";
    save();
    render();
    toast("Room deleted", "danger");
  }

  /* ---------- Item actions ---------- */
  function deleteItem(id) {
    var item = state.items.find(function (i) { return i.id === id; });
    if (!item) return;
    if (!window.confirm("Delete “" + item.name + "”? This can’t be undone.")) return;
    state.items = state.items.filter(function (i) { return i.id !== id; });
    if (editingItemId === id) resetItemForm();
    save();
    render();
    toast("Item deleted", "danger");
  }

  /* ---------- Event delegation ---------- */
  roomListEl.addEventListener("click", function (e) {
    var btn = e.target.closest("[data-action]");
    if (!btn) return;
    var action = btn.getAttribute("data-action");
    var id = btn.getAttribute("data-id");
    if (action === "filter-room") {
      filters.room = (filters.room === id) ? "all" : id;
      render();
    } else if (action === "rename-room") {
      renameRoom(id);
    } else if (action === "delete-room") {
      deleteRoom(id);
    }
  });

  itemListEl.addEventListener("click", function (e) {
    var btn = e.target.closest("[data-action]");
    if (!btn) return;
    var action = btn.getAttribute("data-action");
    var id = btn.getAttribute("data-id");
    if (action === "edit-item") startEditItem(id);
    else if (action === "delete-item") deleteItem(id);
  });

  /* ---------- Filters ---------- */
  document.getElementById("searchInput").addEventListener("input", function (e) {
    filters.query = e.target.value;
    render();
  });
  filterRoomEl.addEventListener("change", function (e) { filters.room = e.target.value; render(); });
  filterCategoryEl.addEventListener("change", function (e) { filters.category = e.target.value; render(); });
  clearFiltersEl.addEventListener("click", function () {
    filters = { query: "", room: "all", category: "all" };
    document.getElementById("searchInput").value = "";
    render();
  });

  /* ---------- Export / Import / Demo ---------- */
  document.getElementById("btnExport").addEventListener("click", function () {
    var payload = { app: "home-inventory", version: 1, exportedAt: new Date().toISOString(), rooms: state.rooms, items: state.items };
    var blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });
    var url = URL.createObjectURL(blob);
    var a = document.createElement("a");
    a.href = url;
    a.download = "home-inventory-" + new Date().toISOString().slice(0, 10) + ".json";
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(function () { URL.revokeObjectURL(url); }, 1000);
    toast("Inventory exported", "success");
  });

  var importFile = document.getElementById("importFile");
  document.getElementById("btnImport").addEventListener("click", function () { importFile.click(); });
  importFile.addEventListener("change", function () {
    var file = importFile.files && importFile.files[0];
    if (!file) return;
    var reader = new FileReader();
    reader.onload = function () {
      try {
        var data = JSON.parse(reader.result);
        if (!data || !Array.isArray(data.rooms) || !Array.isArray(data.items)) {
          throw new Error("bad shape");
        }
        var rooms = data.rooms.map(function (r) {
          return { id: String(r.id || uid()), name: String(r.name || "Room") };
        });
        var roomIds = {};
        rooms.forEach(function (r) { roomIds[r.id] = true; });
        var items = data.items.map(function (i) {
          var roomId = (i.roomId && roomIds[i.roomId]) ? i.roomId : (rooms[0] ? rooms[0].id : "");
          return {
            id: String(i.id || uid()),
            name: String(i.name || "Item"),
            quantity: Math.max(0, Math.floor(Number(i.quantity) || 0)),
            category: CATEGORIES.indexOf(i.category) !== -1 ? i.category : "Other",
            roomId: roomId,
            notes: String(i.notes || "")
          };
        });
        state = { rooms: rooms, items: items };
        filters = { query: "", room: "all", category: "all" };
        document.getElementById("searchInput").value = "";
        save();
        render();
        toast("Imported " + items.length + " items", "success");
      } catch (err) {
        toast("That file isn’t a valid inventory", "danger");
      }
      importFile.value = "";
    };
    reader.onerror = function () { toast("Could not read the file", "danger"); importFile.value = ""; };
    reader.readAsText(file);
  });

  function loadDemo() {
    if (state.items.length > 0 && !window.confirm("Replace your current data with the demo data?")) return;
    state = demoData();
    filters = { query: "", room: "all", category: "all" };
    document.getElementById("searchInput").value = "";
    save();
    render();
    toast("Demo data loaded", "success");
  }
  document.getElementById("btnDemo").addEventListener("click", loadDemo);
  emptyActionEl.addEventListener("click", loadDemo);

  /* ---------- Init ---------- */
  load();
  renderCategorySelects();
  resetItemForm();
  render();
})();
