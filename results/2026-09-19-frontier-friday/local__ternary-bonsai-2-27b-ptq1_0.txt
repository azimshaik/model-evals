(function () {
  "use strict";

  const STORAGE_KEY = "homeInventory.v1";
  const CATEGORIES = ["Food", "Tools", "Documents", "Electronics", "Cleaning", "Other"];
  const LOW_STOCK_THRESHOLD = 2;

  // ---------- State ----------
  let state = { rooms: [], items: [] };
  let editingItemId = null;
  let selectedRoomId = null;

  // ---------- DOM refs ----------
  const $ = (id) => document.getElementById(id);
  const roomListEl = $("roomList");
  const newRoomNameEl = $("newRoomName");
  const renameRoomNameEl = $("renameRoomName");
  const deleteRoomNameEl = $("deleteRoomName");
  const btnAddRoom = $("btnAddRoom");
  const btnRenameRoom = $("btnRenameRoom");
  const btnDeleteRoom = $("btnDeleteRoom");
  const searchInput = $("searchInput");
  const roomFilter = $("roomFilter");
  const categoryFilter = $("categoryFilter");
  const itemForm = $("itemForm");
  const itemNameEl = $("itemName");
  const itemQtyEl = $("itemQty");
  const itemCategoryEl = $("itemCategory");
  const itemRoomEl = $("itemRoom");
  const itemNotesEl = $("itemNotes");
  const btnSaveItem = $("btnSaveItem");
  const btnCancelEdit = $("btnCancelEdit");
  const itemListEl = $("itemList");
  const emptyStateEl = $("emptyState");
  const statsBarEl = $("statsBar");
  const toastEl = $("toast");
  const btnDemo = $("btnDemo");
  const btnExport = $("btnExport");
  const btnImport = $("btnImport");
  const fileInput = $("fileInput");

  // ---------- Utilities ----------
  function uid() {
    return Date.now().toString(36) + Math.random().toString(36).slice(2, 9);
  }

  function escapeHtml(str) {
    const div = document.createElement("div");
    div.textContent = str;
    return div.innerHTML;
  }

  function showToast(message, type) {
    toastEl.textContent = message;
    toastEl.className = "toast " + (type || "success");
    requestAnimationFrame(() => {
      toastEl.classList.add("show");
    });
    clearTimeout(toastEl._timer);
    toastEl._timer = setTimeout(() => {
      toastEl.classList.remove("show");
    }, 2200);
  }

  // ---------- Persistence ----------
  function saveState() {
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
    } catch (e) {
      showToast("Could not save data", "error");
    }
  }

  function loadState() {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (raw) {
        const parsed = JSON.parse(raw);
        if (parsed && Array.isArray(parsed.rooms) && Array.isArray(parsed.items)) {
          state = parsed;
        }
      }
    } catch (e) {
      state = { rooms: [], items: [] };
    }
  }

  // ---------- Data helpers ----------
  function getRoomById(id) {
    return state.rooms.find((r) => r.id === id) || null;
  }

  function getRoomName(id) {
    return getRoomById(id) ? getRoomById(id).name : "No room";
  }

  function countItemsInRoom(roomId) {
    return state.items.filter((i) => i.roomId === roomId).length;
  }

  function isLowStock(item) {
    return Number(item.quantity) <= LOW_STOCK_THRESHOLD;
  }

  function findItemsInRoom(roomId) {
    return state.items.filter((i) => i.roomId === roomId);
  }

  // ---------- Rendering ----------
  function renderRooms() {
    roomListEl.innerHTML = "";
    state.rooms.forEach((room) => {
      const li = document.createElement("li");
      li.className = "room-chip";
      li.setAttribute("tabindex", "0");
      li.setAttribute("role", "button");
      li.setAttribute("aria-label", "Room: " + room.name + ". Press Enter to select.");
      li.dataset.roomId = room.id;

      const nameSpan = document.createElement("span");
      nameSpan.className = "room-name";
      nameSpan.textContent = room.name;

      const countSpan = document.createElement("span");
      countSpan.className = "room-count";
      countSpan.textContent = countItemsInRoom(room.id);

      const actions = document.createElement("span");
      actions.className = "room-actions";

      const editBtn = document.createElement("button");
      editBtn.type = "button";
      editBtn.className = "icon-btn";
      editBtn.setAttribute("aria-label", "Edit room " + room.name);
      editBtn.textContent = "✏️";
      editBtn.addEventListener("click", () => selectRoomForEdit(room.id));

      const delBtn = document.createElement("button");
      delBtn.type = "button";
      delBtn.className = "icon-btn danger";
      delBtn.setAttribute("aria-label", "Delete room " + room.name);
      delBtn.textContent = "🗑";
      delBtn.addEventListener("click", () => confirmDeleteRoom(room.id));

      actions.appendChild(editBtn);
      actions.appendChild(delBtn);

      li.appendChild(nameSpan);
      li.appendChild(countSpan);
      li.appendChild(actions);
      li.addEventListener("click", () => selectRoom(room.id));
      li.addEventListener("keydown", (e) => {
        if (e.key === "Enter" || e.key === " ") {
          e.preventDefault();
          selectRoom(room.id);
        }
      });

      roomListEl.appendChild(li);
    });

    if (state.rooms.length === 0) {
      const li = document.createElement("li");
      li.className = "empty-state";
      li.textContent = "No rooms yet. Add one above.";
      roomListEl.appendChild(li);
    }
  }

  function renderFilters() {
    // Room filter
    roomFilter.innerHTML = "";
    const allOpt = document.createElement("option");
    allOpt.value = "";
    allOpt.textContent = "All rooms";
    roomFilter.appendChild(allOpt);
    state.rooms.forEach((room) => {
      const opt = document.createElement("option");
      opt.value = room.id;
      opt.textContent = room.name;
      roomFilter.appendChild(opt);
    });

    // Category filter
    categoryFilter.innerHTML = "";
    const catAll = document.createElement("option");
    catAll.value = "";
    catAll.textContent = "All categories";
    categoryFilter.appendChild(catAll);
    CATEGORIES.forEach((cat) => {
      const opt = document.createElement("option");
      opt.value = cat;
      opt.textContent = cat;
      categoryFilter.appendChild(opt);
    });

    // Item form room select
    itemRoomEl.innerHTML = "";
    const noRoom = document.createElement("option");
    noRoom.value = "";
    noRoom.textContent = "Choose a room…";
    itemRoomEl.appendChild(noRoom);
    state.rooms.forEach((room) => {
      const opt = document.createElement("option");
      opt.value = room.id;
      opt.textContent = room.name;
      itemRoomEl.appendChild(opt);
    });

    // Item form category select
    itemCategoryEl.innerHTML = "";
    CATEGORIES.forEach((cat) => {
      const opt = document.createElement("option");
      opt.value = cat;
      opt.textContent = cat;
      itemCategoryEl.appendChild(opt);
    });
  }

  function renderItems() {
    const query = searchInput.value.trim().toLowerCase();
    const filterRoom = roomFilter.value;
    const filterCat = categoryFilter.value;

    let filtered = state.items.slice();
    if (query) {
      filtered = filtered.filter((i) =>
        i.name.toLowerCase().includes(query) ||
        (i.notes && i.notes.toLowerCase().includes(query))
      );
    }
    if (filterRoom) {
      filtered = filtered.filter((i) => i.roomId === filterRoom);
    }
    if (filterCat) {
      filtered = filtered.filter((i) => i.category === filterCat);
    }

    itemListEl.innerHTML = "";
    filtered.forEach((item) => {
      const li = document.createElement("li");
      const card = document.createElement("div");
      card.className = "item-card" + (isLowStock(item) ? " low-stock" : "");

      const main = document.createElement("div");
      main.className = "item-main";

      const nameEl = document.createElement("div");
      nameEl.className = "item-name";
      nameEl.textContent = item.name;

      const meta = document.createElement("div");
      meta.className = "item-meta";

      const qtyBadge = document.createElement("span");
      qtyBadge.className = "badge" + (isLowStock(item) ? " low" : "");
      qtyBadge.textContent = isLowStock(item) ? "⚠ Low: " + item.quantity : "Qty: " + item.quantity;

      const catBadge = document.createElement("span");
      catBadge.className = "badge";
      catBadge.textContent = item.category;

      const roomBadge = document.createElement("span");
      roomBadge.className = "badge";
      roomBadge.textContent = getRoomName(item.roomId);

      meta.appendChild(qtyBadge);
      meta.appendChild(catBadge);
      meta.appendChild(roomBadge);

      main.appendChild(nameEl);
      main.appendChild(meta);

      if (item.notes) {
        const notesEl = document.createElement("div");
        notesEl.className = "item-notes";
        notesEl.textContent = item.notes;
        main.appendChild(notesEl);
      }

      const actions = document.createElement("div");
      actions.className = "item-actions";

      const editBtn = document.createElement("button");
      editBtn.type = "button";
      editBtn.className = "icon-btn";
      editBtn.setAttribute("aria-label", "Edit item " + item.name);
      editBtn.textContent = "✏️";
      editBtn.addEventListener("click", () => startEditItem(item.id));

      const delBtn = document.createElement("button");
      delBtn.type = "button";
      delBtn.className = "icon-btn danger";
      delBtn.setAttribute("aria-label", "Delete item " + item.name);
      delBtn.textContent = "🗑";
      delBtn.addEventListener("click", () => confirmDeleteItem(item.id));

      actions.appendChild(editBtn);
      actions.appendChild(delBtn);

      card.appendChild(main);
      card.appendChild(actions);
      li.appendChild(card);
      itemListEl.appendChild(li);
    });

    emptyStateEl.classList.toggle("hidden", filtered.length > 0);
    renderStats(filtered);
  }

  function renderStats(filtered) {
    const total = state.items.length;
    const low = state.items.filter(isLowStock).length;
    const rooms = state.rooms.length;
    statsBarEl.innerHTML = "";
    const stats = [
      { label: "Total items", value: total },
      { label: "Low stock", value: low },
      { label: "Rooms", value: rooms },
      { label: "Showing", value: filtered.length }
    ];
    stats.forEach((s) => {
      const div = document.createElement("div");
      div.className = "stat";
      div.innerHTML = s.label + ": <strong>" + s.value + "</strong>";
      statsBarEl.appendChild(div);
    });
  }

  function renderAll() {
    renderRooms();
    renderFilters();
    renderItems();
  }

  // ---------- Room actions ----------
  function addRoom() {
    const name = newRoomNameEl.value.trim();
    if (!name) {
      showToast("Please enter a room name", "error");
      newRoomNameEl.focus();
      return;
    }
    const existing = state.rooms.find((r) => r.name.toLowerCase() === name.toLowerCase());
    if (existing) {
      showToast("A room named \"" + name + "\" already exists", "error");
      return;
    }
    state.rooms.push({ id: uid(), name: name });
    newRoomNameEl.value = "";
    saveState();
    renderAll();
    showToast("Room \"" + name + "\" added", "success");
  }

  function selectRoom(id) {
    selectedRoomId = id;
    renameRoomNameEl.value = getRoomName(id);
    deleteRoomNameEl.value = getRoomName(id);
    newRoomNameEl.focus();
  }

  function selectRoomForEdit(id) {
    selectedRoomId = id;
    renameRoomNameEl.value = getRoomName(id);
    renameRoomNameEl.focus();
  }

  function renameRoom() {
    if (!selectedRoomId) {
      showToast("Select a room to rename first", "error");
      return;
    }
    const name = renameRoomNameEl.value.trim();
    if (!name) {
      showToast("Please enter a new room name", "error");
      return;
    }
    const room = getRoomById(selectedRoomId);
    if (!room) return;
    const existing = state.rooms.find((r) => r.id !== selectedRoomId && r.name.toLowerCase() === name.toLowerCase());
    if (existing) {
      showToast("A room named \"" + name + "\" already exists", "error");
      return;
    }
    room.name = name;
    saveState();
    renderAll();
    showToast("Room renamed to \"" + name + "\"", "success");
  }

  function confirmDeleteRoom(id) {
    const room = getRoomById(id);
    if (!room) return;
    const count = countItemsInRoom(id);
    if (count > 0) {
      if (!confirm("Room \"" + room.name + "\" has " + count + " item(s). Delete the room and its items?")) {
        return;
      }
    } else {
      if (!confirm("Delete room \"" + room.name + "\"?")) {
        return;
      }
    }
    state.rooms = state.rooms.filter((r) => r.id !== id);
    state.items = state.items.filter((i) => i.roomId !== id);
    if (selectedRoomId === id) selectedRoomId = null;
    saveState();
    renderAll();
    showToast("Room \"" + room.name + "\" deleted", "success");
  }

  // ---------- Item actions ----------
  function startEditItem(id) {
    const item = state.items.find((i) => i.id === id);
    if (!item) return;
    editingItemId = id;
    itemNameEl.value = item.name;
    itemQtyEl.value = item.quantity;
    itemCategoryEl.value = item.category;
    itemRoomEl.value = item.roomId;
    itemNotesEl.value = item.notes || "";
    btnSaveItem.textContent = "Update item";
    btnCancelEdit.hidden = false;
    itemForm.scrollIntoView({ behavior: "smooth", block: "center" });
    itemNameEl.focus();
  }

  function cancelEdit() {
    editingItemId = null;
    btnSaveItem.textContent = "Save item";
    btnCancelEdit.hidden = true;
    itemForm.reset();
    itemQtyEl.value = "1";
    itemCategoryEl.value = CATEGORIES[0];
    itemRoomEl.value = "";
    itemNotesEl.value = "";
    renderFilters();
  }

  function saveItem() {
    const name = itemNameEl.value.trim();
    const qty = parseInt(itemQtyEl.value, 10);
    const category = itemCategoryEl.value;
    const roomId = itemRoomEl.value;
    const notes = itemNotesEl.value.trim();

    if (!name) {
      showToast("Item name is required", "error");
      itemNameEl.focus();
      return;
    }
    if (isNaN(qty) || qty < 0) {
      showToast("Quantity must be a non-negative number", "error");
      itemQtyEl.focus();
      return;
    }
    if (!category) {
      showToast("Please choose a category", "error");
      itemCategoryEl.focus();
      return;
    }
    if (!roomId) {
      showToast("Please choose a room", "error");
      itemRoomEl.focus();
      return;
    }

    if (editingItemId) {
      const item = state.items.find((i) => i.id === editingItemId);
      if (item) {
        item.name = name;
        item.quantity = qty;
        item.category = category;
        item.roomId = roomId;
        item.notes = notes;
      }
      editingItemId = null;
      btnSaveItem.textContent = "Save item";
      btnCancelEdit.hidden = true;
      showToast("Item updated", "success");
    } else {
      state.items.push({
        id: uid(),
        name: name,
        quantity: qty,
        category: category,
        roomId: roomId,
        notes: notes
      });
      showToast("Item added", "success");
    }

    itemForm.reset();
    itemQtyEl.value = "1";
    itemCategoryEl.value = CATEGORIES[0];
    itemRoomEl.value = "";
    itemNotesEl.value = "";
    renderFilters();
    saveState();
    renderItems();
  }

  function confirmDeleteItem(id) {
    const item = state.items.find((i) => i.id === id);
    if (!item) return;
    if (!confirm("Delete item \"" + item.name + "\"?")) {
      return;
    }
    state.items = state.items.filter((i) => i.id !== id);
    if (editingItemId === id) {
      cancelEdit();
    }
    saveState();
    renderItems();
    showToast("Item deleted", "success");
  }

  // ---------- Export / Import ----------
  function exportData() {
    const blob = new Blob([JSON.stringify(state, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = "home-inventory-" + new Date().toISOString().slice(0, 10) + ".json";
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
    showToast("Data exported", "success");
  }

  function importData() {
    fileInput.value = "";
    fileInput.addEventListener("change", handleFileImport, { once: true });
    fileInput.click();
  }

  function handleFileImport(e) {
    const file = e.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = function () {
      try {
        const parsed = JSON.parse(reader.result);
        if (!parsed || !Array.isArray(parsed.rooms) || !Array.isArray(parsed.items)) {
          throw new Error("Invalid data format");
        }
        // Basic validation
        if (parsed.rooms.some((r) => !r.id || !r.name)) throw new Error("Invalid room data");
        if (parsed.items.some((i) => !i.id || !i.name || i.quantity === undefined)) {
          throw new Error("Invalid item data");
        }
        if (!confirm("Replace current data with the imported file?")) {
          return;
        }
        state = parsed;
        editingItemId = null;
        selectedRoomId = null;
        saveState();
        renderAll();
        showToast("Data imported successfully", "success");
      } catch (err) {
        showToast("Import failed: " + err.message, "error");
      }
    };
    reader.onerror = function () {
      showToast("Could not read file", "error");
    };
    reader.readAsText(file);
  }
